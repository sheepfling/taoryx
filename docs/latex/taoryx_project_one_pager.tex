\vspace{0.34em}
{\large\color{navy}\bfseries What it is}\par
\vspace{0.18em}
{\small\color{slate}
  \taoryx{} turns the 1995 \emph{Trajectory Analysis and Optimization System}
  (TAOS) User's Manual into a living engineering asset: a rebuildable,
  provenance-linked reference edition plus a typed software platform for
  trajectory modeling, analysis, and mission workflows. It preserves what is
  documented, makes supported behavior inspectable, and leaves unverified
  claims visibly bounded.
}\par

\begin{minipage}[t]{0.48\linewidth}
  \sectiontitle{1. Models, data, and fidelity}
  \fcolorbox{line}{mist}{%
    \begin{minipage}{\dimexpr\linewidth-2\fboxsep-2\fboxrule\relax}
      {\color{navy}\small\bfseries Developer path}\par\vspace{0.14em}
      {\scriptsize\color{slate}\texttt{bootstrap} and select the developer
      profile \(\rightarrow\) \texttt{model list}/\texttt{assess}
      \(\rightarrow\) \texttt{model plan} \(\rightarrow\)
      \texttt{scaffold} \(\rightarrow\) validate/compile \(\rightarrow\)
      batch run, live session, or registered \texttt{model tune}.}\par\vspace{0.14em}
      {\scriptsize\color{slate}The selected plug-in owns its data, adapter,
      controls, focused tests, and evidence. A model can be discoverable while
      honestly advertising a blocked or batch-only realization.}
    \end{minipage}%
  }%\par\vspace{0.34em}

  \scopebox{Evidence-backed data forms}{Legacy \texttt{.tbl}/\texttt{.prb}
  files; DAVE-ML and CADAC inputs; source tables; geometry, mass/inertia,
  aerodynamic, propulsion, actuator, atmosphere/wind, route, waypoint, and
  telemetry data. Typed metadata retains units, frames, bounds, provenance,
  and availability.}\par\vspace{0.34em}

  \scopebox{Existing model portfolio}{Analytical Simple Aero; A320, F-16,
  Hummingbird, X-15, HL-20, Skywalker X8, B747, NESC staging/replay, passive
  bodies, dual launch, and parametric interceptors. Reachability is an optional
  overlay; CADAC is a separately installable source-bound catalog.}\par\vspace{0.34em}

  \fcolorbox{line}{white}{%
    \begin{minipage}{\dimexpr\linewidth-2\fboxsep-2\fboxrule\relax}
      {\color{teal}\small\bfseries Fidelity ladder}\par\vspace{0.13em}
      {\scriptsize\color{slate}\textbf{1. Point-mass 3DOF:} translation only.
      \textbf{2. Pseudo-6DOF:} declared attitude/rate response bridge.
      \textbf{3. Rigid-body direct wrench:} full Newton-Euler dynamics without
      an effector claim. \textbf{4. Surface allocated:} declared effectors,
      limits, and allocation.}\par\vspace{0.2em}
      {\scriptsize\color{slate}\textbf{Important:} fidelity is an explicit
      request graph, not a silent switch. Step-up always requests a named tier;
      step-down is either exact-only or an opt-in, evidence-qualified
      \texttt{validated\_lower\_only} fallback.}
    \end{minipage}%
  }%
\end{minipage}%

\begin{minipage}[t]{0.495\linewidth}
  \sectiontitle{2. Vehicle Composition API}
  \fcolorbox{line}{mist}{%
    \begin{minipage}{\dimexpr\linewidth-2\fboxsep-2\fboxrule\relax}
      {\color{navy}\small\bfseries One generic front door for every model}\par\vspace{0.16em}
      {\scriptsize\color{slate}\texttt{list\_models()} \(\rightarrow\)
      choose model, realization, fidelity, and operation \(\rightarrow\)
      fetch typed configuration and output schemas \(\rightarrow\)
      validate a fingerprinted configuration \(\rightarrow\) batch trajectory
      or persistent \texttt{open / inspect / step / reset / close} session.}
    \end{minipage}%
  }%\par\vspace{0.34em}

  \scopebox{Built for UI, AI, and library consumers}{The provider publishes
  labels, units, bounds, choices, defaults, validation, compatible fidelity,
  available telemetry, and control authority. A client builds forms and model
  pickers from this publication without importing a vehicle-specific dynamics
  class or guessing action meanings from names.}\par\vspace{0.34em}

  \fcolorbox{line}{white}{%
    \begin{minipage}{\dimexpr\linewidth-2\fboxsep-2\fboxrule\relax}
      {\color{teal}\small\bfseries HTML / JavaScript opportunity}\par\vspace{0.13em}
      {\scriptsize\color{slate}Schemas and diagnostics are JSON-safe,
      fingerprinted, unit-aware, and intended for generated forms, catalogues,
      plots, and agent clients. Today the public contract is transport-neutral
      Python/domain API, not an included REST or WebSocket service. A thin web
      adapter can expose discovery, configuration, batch results, and session
      steps while the application owns authentication, reconnect, and deadman
      safety policy.}
    \end{minipage}%
  }%\par\vspace{0.34em}

  \sectiontitle{3. Controls: intent to plant}
  \fcolorbox{line}{mist}{%
    \begin{minipage}{\dimexpr\linewidth-2\fboxsep-2\fboxrule\relax}
      {\scriptsize\color{slate}\textbf{High level:} provider/open-loop,
      destination, waypoint, route. \textbf{Reduced:} position, velocity,
      flight path, energy. \textbf{Operator:} pilot-like axes, attitude, or
      body rate. \textbf{Expert/diagnostic:} direct wrench or direct effector.}\par\vspace{0.22em}
      {\scriptsize\color{slate}\textbf{Lowering chain:} mission intent
      \(\rightarrow\) guidance/reference \(\rightarrow\) regulator
      (LQR/LQI or declared family control) \(\rightarrow\) generalized demand
      \(\rightarrow\) allocator and actuator limits \(\rightarrow\) plant.}\par\vspace{0.22em}
      {\scriptsize\color{slate}A session selects one coherent authority
      profile at a time, not a misleading union of every model control. Each
      step returns requested, applied, achieved, lowered, and availability
      feedback so a consumer can see what really reached the model.}
    \end{minipage}%
  }%
\end{minipage}
